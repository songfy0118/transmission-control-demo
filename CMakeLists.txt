cmake_minimum_required(VERSION 3.16)
project(transmission_control_demo LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(speed_pid src/pid.c)
target_include_directories(speed_pid PUBLIC include)
if(NOT MSVC)
  target_link_libraries(speed_pid PUBLIC m)
  target_compile_options(speed_pid PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(simulate src/simulate.cpp)
target_link_libraries(simulate PRIVATE speed_pid)
add_executable(control_tests tests/control_tests.cpp)
target_link_libraries(control_tests PRIVATE speed_pid)
if(NOT MSVC)
  target_compile_options(simulate PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(control_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME control_tests COMMAND control_tests)
